const hasTokenExpired = () => {
  const now = Date.now();
  // Should expire 10 seconds before actual expiry time
  const expiry = accessTokenExpiry.getTime() - 10_000;
  return now >= expiry;
};

  async (config) => {
    if (hasTokenExpired()) {
      const result = await axios.get(accessTokenUrl);
      accessToken = result.data.Token;
      accessTokenExpiry = new Date(result.data.TokenExpiration);
    }

    config.headers.Authorization = `Bearer ${accessToken}`;
    config.headers["X-XSRF-Token"] = reqToken;

    return config;
  },
  (error) => Promise.reject(error),
);

export const formatCurrency = (amount: number): string => {
  if (typeof amount !== "number") {
    throw new Error("Invalid argument: amount must be a number");
  }
  return new Intl.NumberFormat("en-GB", { style: "currency", currency: "GBP" })
    .format(amount)
    .slice(1);
};

export const formatSortCodeString = (sortcode: string) => {
  if (typeof sortcode !== "string") {
    throw new Error("Invalid argument: sortcode must be string");
  }
  const codeStr = String(sortcode).padStart(6, "0");
  return `${codeStr.slice(0, 2)}-${codeStr.slice(2, 4)}-${codeStr.slice(4, 6)}`;
};


import { formatSortCodeString } from "./formatSortCodeString";

  });

  test("pads and formats a sort code with less than 6 digits", () => {
    const result = formatSortCodeString("123");
    expect(result).toBe("00-01-23");
  });

  test("throws an error if the argument is not a string", () => {
    expect(() => formatSortCodeString(123456 as any)).toThrow(
      "Invalid argument: sortcode must be string",
    );
  });

  test("pads a sort code of length 5 with leading zeros", () => {
    const result = formatSortCodeString("12345");
    expect(result).toBe("01-23-45");
  });

  test('formats an empty string to "00-00-00"', () => {
    const result = formatSortCodeString("");
    expect(result).toBe("00-00-00");
  });

  test("pads a sort code of length 1 with leading zeros", () => {

export const formatDateString = (dateString: string): string => {
  const date = new Date(dateString);
  const options: Intl.DateTimeFormatOptions = {
    year: "numeric",
    month: "short",
    day: "numeric",
  };
  return date.toLocaleDateString("en-US", options);
};
